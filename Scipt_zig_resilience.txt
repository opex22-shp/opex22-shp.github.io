// ===================== 2. CHALEUR : LANDSAT 8/9 =====================
function maskL(img) {
  var qa = img.select('QA_PIXEL');
  var m = qa.bitwiseAnd(1 << 1).eq(0)   // nuage dilate
    .and(qa.bitwiseAnd(1 << 2).eq(0))   // cirrus
    .and(qa.bitwiseAnd(1 << 3).eq(0))   // nuage
    .and(qa.bitwiseAnd(1 << 4).eq(0));  // ombre de nuage
  return img.updateMask(m);
}
function prepL(img) {
  var lst = img.select('ST_B10').multiply(0.00341802).add(149.0)
    .subtract(273.15).rename('lst');                       // Kelvin -> degres C
  var red = img.select('SR_B4').multiply(0.0000275).add(-0.2);
  var nir = img.select('SR_B5').multiply(0.0000275).add(-0.2);
  var ndvi = nir.subtract(red).divide(nir.add(red)).rename('ndvi_l');
  return lst.addBands(ndvi);
}
function landsatCol(y0, y1) {
  return ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
    .merge(ee.ImageCollection('LANDSAT/LC09/C02/T1_L2'))
    .filterBounds(aoi)
    .filterDate(y0 + '-01-01', y1 + '-12-31')
    .filter(ee.Filter.calendarRange(MONTHS[0], MONTHS[1], 'month'))
    .filter(ee.Filter.lt('CLOUD_COVER', 40))
    .map(maskL).map(prepL);
}
function landsatComp(y0, y1) {
  return landsatCol(y0, y1).median().clip(aoi);
}
print('Nb images Landsat (periode courante)', landsatCol(YEAR_START, YEAR_END).size());
var lstNow = landsatComp(YEAR_START, YEAR_END).select('lst');

// ===================== 3. VEGETATION, BATI, EAU =====================
function maskS2(img) {
  var scl = img.select('SCL');
  var ok = scl.neq(3).and(scl.neq(8)).and(scl.neq(9)).and(scl.neq(10)).and(scl.neq(11));
  return img.updateMask(ok);
}
var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(aoi)
  .filterDate(YEAR_START + '-01-01', YEAR_END + '-12-31')
  .filter(ee.Filter.calendarRange(MONTHS[0], MONTHS[1], 'month'))
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 30))
  .map(maskS2);
print('Nb images Sentinel-2', s2.size());
var s2c = s2.median().clip(aoi);
var ndvi = s2c.normalizedDifference(['B8', 'B4']).rename('ndvi');

var built = wc.eq(50).rename('built');   // classe 50 = zone baties

  {min: 0, max: 0.8, palette: ['ffffff', 'ffff66', '1a9850']}, 'NDVI', false);

// ===================== 4. MAILLES + INDICATEURS =====================
var grid = aoi.coveringGrid(ee.Projection('EPSG:32628'), GRID_SIZE);
var stack = lstNow.addBands(ndvi).addBands(built).updateMask(land);

var cells = stack.reduceRegions({
  collection: grid, reducer: ee.Reducer.mean(), scale: 10, tileScale: 4
}).filter(ee.Filter.notNull(['lst', 'ndvi', 'built']));

// Population (WorldPop, estimation modelisee 2020)
var pop = ee.ImageCollection('WorldPop/GP/100m/pop')
  .filter(ee.Filter.eq('country', 'SEN'))
  .filter(ee.Filter.eq('year', 2020))
  .mosaic().select('population');
cells = pop.reduceRegions({
  collection: cells, reducer: ee.Reducer.sum().setOutputs(['pop']), scale: 100
});

// Equipements sensibles (ecoles, marches, structures de sante)
if (USE_EQUIP) {
  var equip = ee.FeatureCollection(EQUIP_ASSET);
  cells = cells.map(function (c) {
    var e = equip.filterBounds(c.geometry());
    var n = function (t) { return e.filter(ee.Filter.eq('type', t)).size(); };
    return c.set({ecole: n('education'), sante: n('sante'), marche: n('marche')});
  });
} else {
  cells = cells.map(function (c) {
    return c.set({ecole: 0, sante: 0, marche: 0});
  });
}
cells = cells.map(function (c) {
  var eq = ee.Number(c.get('ecole')).add(ee.Number(c.get('marche')))
    .add(ee.Number(c.get('sante')).multiply(2));       // sante pondere x2
  return c.set('equip', eq);
});

// Anomalie thermique = ecart a la moyenne des mailles de terre ferme
var lstMean = cells.aggregate_mean('lst').getInfo();
print('LST moyenne de la zone (C)', lstMean);
cells = cells.map(function (c) {
  return c.set('anom', ee.Number(c.get('lst')).subtract(lstMean));
});

// ===================== 5. SCORE DE PRIORITE =====================
function rng(prop) {
  var d = cells.reduceColumns(ee.Reducer.percentile([5, 95]), [prop]).getInfo();
  return {lo: d.p5, hi: d.p95};
}
var R = {lst: rng('lst'), ndvi: rng('ndvi'), built: rng('built'),
         pop: rng('pop'), equip: rng('equip')};
function norm(c, prop, inv) {
  var r = R[prop];
  var n = ee.Number(c.get(prop)).subtract(r.lo)
    .divide(Math.max(r.hi - r.lo, 1e-9)).max(0).min(1);
  return inv ? ee.Number(1).subtract(n) : n;
}
var eqW = USE_EQUIP ? 0.5 : 0;
var scored = cells.map(function (c) {
  var heat = norm(c, 'lst', false);
  var veg  = norm(c, 'ndvi', true);     // peu de vegetation = score eleve
  var imp  = norm(c, 'built', false);
  var expo = norm(c, 'pop', false).multiply(1 - eqW)
    .add(norm(c, 'equip', false).multiply(eqW));
  var score = heat.multiply(W.heat).add(veg.multiply(W.veg))
    .add(imp.multiply(W.imp)).add(expo.multiply(W.expo)).multiply(100);
  return c.set({n_heat: heat, n_veg: veg, n_imp: imp, n_expo: expo, score: score});
});

// INTERACTIVITÉ AU CLIC & FORMATAGE DES RÉSULTATS
// =========================================================================

Map.onClick(function (coords) {
  info.clear();
  info.add(ui.Label('Chargement des données...', {fontStyle: 'italic', color: '#9ca3af'}));
  
  var pt = ee.Geometry.Point([coords.lon, coords.lat]);
  quartierLabel.setValue('Recherche du quartier...');
  
  quartiers.filterBounds(pt).limit(1).evaluate(function (q) {
    quartierLabel.setValue(q && q.features.length > 0
      ? 'Quartier : ' + q.features[0].properties[QUARTIER_FIELD]
      : 'Quartier : Hors commune / Non identifié');
  });
  
  scored.filterBounds(pt).limit(1).evaluate(function (fc) {
    info.clear();
    if (!fc || fc.features.length === 0) { 
      info.add(ui.Label('Aucune donnée disponible sur cette zone.', {color: '#dc2626'})); 
      return; 
    }
    
    var p = fc.features[0].properties;
    var lvl = p.score >= 75 ? 'TRÈS ÉLEVÉE' : p.score >= 50 ? 'ÉLEVÉE' : p.score >= 25 ? 'MODÉRÉE' : 'FAIBLE';
    var lvlColor = p.score >= 75 ? '#dc2626' : p.score >= 50 ? '#f97316' : p.score >= 25 ? '#eab308' : '#16a34a';
    
    var rec = [];
    if (p.n_veg >= 0.66) rec.push('• Végétalisation ciblée (micro-forêts)');
    if (p.n_imp >= 0.66) rec.push('• Désimperméabilisation et revêtements réfléchissants');
    if (p.n_heat >= 0.66 && (p.ecole + p.marche + p.sante) > 0) rec.push('• Ombrage prioritaire près des équipements publics');
    if (rec.length === 0) rec.push('• Surveillance et maintien de la trame verte');

    // Section 1 : Score de Priorité
    info.add(ui.Panel([
      ui.Label('NIVEAU D\'URGENCE : ' + lvl, {fontWeight: 'bold', fontSize: '13px', color: lvlColor}),
      ui.Label('Score global : ' + p.score.toFixed(0) + '/100', {fontSize: '12px', color: '#6b7280', margin: '2px 0 10px 0'})
    ], null, {backgroundColor: '#f9fafb', padding: '10px', border: '1px solid #e5e7eb', borderRadius: '6px'}));

    // Section 2 : Indicateurs Physiques & Socio-démographiques
    var gridPanel = ui.Panel({
      layout: ui.Panel.Layout.Flow('vertical'),
      style: {margin: '12px 0', backgroundColor: '#ffffff'}
    });
    
    gridPanel.add(ui.Label('Données Environnementales :', {fontWeight: 'bold', fontSize: '12px', color: '#374151'}));
    gridPanel.add(ui.Label('• Température de surface : ' + p.lst.toFixed(1) + ' °C (Anomalie : ' + (p.anom >= 0 ? '+' : '') + p.anom.toFixed(1) + ' °C)', {fontSize: '12px', color: '#4b5563'}));
    gridPanel.add(ui.Label('• Indice de végétation (NDVI) : ' + p.ndvi.toFixed(2) + ' | Densité du bâti : ' + (p.built * 100).toFixed(0) + ' %', {fontSize: '12px', color: '#4b5563'}));
    
    gridPanel.add(ui.Label('Vulnérabilité Sociale :', {fontWeight: 'bold', fontSize: '12px', color: '#374151', margin: '8px 0 2px 0'}));
    gridPanel.add(ui.Label('• Population estimée : ' + Math.round(p.pop) + ' hab. / maille', {fontSize: '12px', color: '#4b5563'}));
    gridPanel.add(ui.Label('• Équipements sensibles : ' + p.ecole + ' école(s), ' + p.sante + ' structure(s) de santé, ' + p.marche + ' marché(s)', {fontSize: '12px', color: '#4b5563'}));
    
    info.add(gridPanel);

    // Section 3 : Actions recommandées
    var recPanel = ui.Panel({style: {backgroundColor: '#eff6ff', padding: '10px', borderRadius: '6px', border: '1px solid #bfdbfe'}});
    recPanel.add(ui.Label('Interventions suggérées :', {fontWeight: 'bold', fontSize: '12px', color: '#1e40af'}));
    rec.forEach(function(r) {
      recPanel.add(ui.Label(r, {fontSize: '12px', color: '#1e3a8a', margin: '3px 0'}));
    });
    info.add(recPanel);
  });
});

  collection: grid, reducer: ee.Reducer.mean(), scale: 30, tileScale: 4
}).filter(ee.Filter.notNull(['dlst', 'dndvi']));
print('Correlation dNDVI / dLST (attendu : negative)',
  chg.reduceColumns(ee.Reducer.pearsonsCorrelation(), ['dndvi', 'dlst']));
print('dLST moyen la ou le NDVI a recule (< -0.05)',
  chg.filter(ee.Filter.lt('dndvi', -0.05)).aggregate_mean('dlst'));
print('dLST moyen la ou le NDVI a augmente (> 0.02)',
  chg.filter(ee.Filter.gt('dndvi', 0.02)).aggregate_mean('dlst'));
print(ui.Chart.feature.byFeature(chg.limit(1500), 'dndvi', ['dlst'])
  .setChartType('ScatterChart')
  .setOptions({title: 'Variation de LST selon la variation de NDVI',
    hAxis: {title: 'Delta NDVI'}, vAxis: {title: 'Delta LST (C)'},
    pointSize: 2, legend: {position: 'none'}, trendlines: {0: {}}}));
